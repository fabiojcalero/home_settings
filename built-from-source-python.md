# Uninstalling Built From Source Python

## To uninstall built from source python, remove the following files and folders:
- /usr/local/bin/python3.X
- /usr/local/bin/pip3.X
- /usr/local/bin/pydoc
- /usr/local/include/python3.X
- /usr/local/lib/libpython3.X.a
- /usr/local/lib/python3.X
- /usr/local/share/man/python3.X.X
- /usr/local/lib/pkgconfig
- /usr/local/bin/idle3.X
- /usr/local/bin/python3.X.d
- any other python files on /usr/local/bin/
